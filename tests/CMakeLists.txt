function(add_kestrel_test target)
    add_executable(${target} ${ARGN})
    target_include_directories(${target} PRIVATE ../third_party/doctest)
    target_link_libraries(${target} PRIVATE kestrel_core)
    kestrel_enable_asan(${target})
    add_test(NAME ${target} COMMAND ${target})
endfunction()

add_kestrel_test(scanner_test scanner_test.cpp)
add_kestrel_test(source_test source_test.cpp)
add_kestrel_test(line_index_test line_index_test.cpp)
add_kestrel_test(timestamp_index_test timestamp_index_test.cpp)
add_kestrel_test(search_controller_test search_controller_test.cpp)
add_kestrel_test(parse_cli_test parse_cli_test.cpp)
add_kestrel_test(config_test config_test.cpp)
